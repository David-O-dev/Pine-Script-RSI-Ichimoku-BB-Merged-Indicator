//Hull Suite
src = close
string modeSwitch = "Hma"
int length = 55
float lengthMult = 1.0
bool useHtf =false
htf = input.timeframe("240", title="Higher timeframe", group = "Hull Suite")
switchColor = true
candleCol = false
visualSwitch = input.bool(true, title="Show as a Band?", group = "Hull Suite")
thicknessSwitch = input.int(1, title="Line Thickness", group = "Hull Suite")
transpSwitch = input.int(40, title="Band Transparency", step=5, group = "Hull Suite")
hullMovingAverage(_src, _length) => ta.wma(2 * ta.wma(_src, _length / 2) - ta.wma(_src, _length), math.round(math.sqrt(_length)))
expHullMovingAverage(_src, _length) => ta.ema(2 * ta.ema(_src, _length / 2) - ta.ema(_src, _length), math.round(math.sqrt(_length)))
triHullMovingAverage(_src, _length) => ta.wma(ta.wma(_src, _length / 3) * 3 - ta.wma(_src, _length / 2) - ta.wma(_src, _length), _length)
modeSelection(mode, _src, _len) =>
    if mode == "Hma"
        hullMovingAverage(_src, _len)
    else if mode == "Ehma"
        expHullMovingAverage(_src, _len)
    else if mode == "Thma"
        triHullMovingAverage(_src, _len / 2)
    else
        na

_hull = modeSelection(modeSwitch, src, int(length * lengthMult))
HULL = useHtf ? request.security(syminfo.tickerid, htf, _hull) : _hull
MHULL = HULL[0]
SHULL = HULL[2]
hullColor = switchColor ? (HULL > HULL[2] ? color.green : color.red) : color.orange
Fi1 = plot(MHULL, title="MHULL", color=hullColor, linewidth=thicknessSwitch)
Fi2 = plot(visualSwitch ? SHULL : na, title="SHULL", color=hullColor, linewidth=thicknessSwitch)
alertcondition(ta.crossover(MHULL, SHULL), title="Hull trending up.", message="Hull trending up.")
alertcondition(ta.crossover(SHULL, MHULL), title="Hull trending down.", message="Hull trending down.")
fill(Fi1, Fi2, title="Band Filler", color=hullColor)
barcolor(color = candleCol ? (switchColor ? hullColor : na) : na)
